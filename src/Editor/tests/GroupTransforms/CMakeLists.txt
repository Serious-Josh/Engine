cmake_minimum_required(VERSION 3.25)
project(GroupTransformTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ENGINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_executable(GroupTransformTests GroupTransformTests.cpp "${ENGINE_ROOT}/src/Scene/Scene.cpp")
target_include_directories(GroupTransformTests PRIVATE "${ENGINE_ROOT}/src")
target_compile_options(GroupTransformTests PRIVATE /EHsc)
enable_testing()
add_test(NAME GroupTransformTests COMMAND GroupTransformTests)
