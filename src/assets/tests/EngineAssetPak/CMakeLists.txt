cmake_minimum_required(VERSION 3.25)
project(EngineAssetPakTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ENGINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_executable(EngineAssetPakTests EngineAssetPakTests.cpp
    ${ENGINE_ROOT}/src/assets/EngineAssetPak.cpp ${ENGINE_ROOT}/src/assets/AssetID.cpp)
target_include_directories(EngineAssetPakTests PRIVATE ${ENGINE_ROOT}/src)
target_compile_definitions(EngineAssetPakTests PRIVATE NOMINMAX)
target_link_libraries(EngineAssetPakTests PRIVATE ole32 bcrypt)
enable_testing()
add_test(NAME EngineAssetPakTests COMMAND EngineAssetPakTests)
